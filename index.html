// ── FORM 3C — Google Apps Script ──────────────────────────────
// Sheet tab must be named: Form3C
// Columns A→I: Date | SlNo | PatientName | Services | Fees |
//              ReceiptDate | PaymentMode | Notes | SubmittedAt
// Auto-sorts entire sheet by ReceiptDate (col F) after every new entry.

// ── Helper: any date value → JS Date object for sorting ──
function toDate_(val) {
  if (!val) return new Date(0);
  if (val instanceof Date) return val;
  var s = String(val).trim();
  // YYYY-MM-DD
  if (/^\d{4}-\d{2}-\d{2}$/.test(s)) return new Date(s);
  // DD/MM/YYYY
  var p = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
  if (p) return new Date(p[3]+'-'+p[2].padStart(2,'0')+'-'+p[1].padStart(2,'0'));
  return new Date(s);
}

// ── Helper: any date value → "YYYY-MM-DD" string ──
function toISO_(val) {
  var d = toDate_(val);
  if (isNaN(d)) return String(val);
  var y = d.getFullYear();
  var m = String(d.getMonth()+1).padStart(2,'0');
  var dd = String(d.getDate()).padStart(2,'0');
  return y+'-'+m+'-'+dd;
}

// ── Sort data rows by ReceiptDate (column F, index 5), ascending ──
function sortByReceiptDate_(sheet) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 3) return; // nothing to sort (only header + 1 row)
  var dataRange = sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn());
  var data = dataRange.getValues();

  data.sort(function(a, b) {
    var da = toDate_(a[5]); // col F = index 5 = ReceiptDate
    var db = toDate_(b[5]);
    if (da < db) return -1;
    if (da > db) return  1;
    // Tie-break: sort by SlNo (col B = index 1)
    return (parseInt(a[1])||0) - (parseInt(b[1])||0);
  });

  dataRange.setValues(data);
}

// ── POST: receive new entry, append, then sort ──
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Form3C");
  var d     = JSON.parse(e.postData.contents);

  sheet.appendRow([
    d.receiptDate,
    d.slNo,
    d.patientName,
    d.uhid         || "",
    d.services,
    d.fees,
    d.paymentNature || "",
    d.paymentMode   || "",
    d.notes         || "",
    new Date().toLocaleString("en-IN")
  ]);

  // Sort entire sheet by receipt date immediately after append
  sortByReceiptDate_(sheet);

  return ContentService
    .createTextOutput(JSON.stringify({status:"success"}))
    .setMimeType(ContentService.MimeType.JSON);
}

// ── GET: fetch serial number or monthly rows ──
function doGet(e) {
  var action   = e.parameter.action;
  var callback = e.parameter.callback;
  var sheet    = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Form3C");
  var rows     = sheet.getDataRange().getValues();

  function respond(obj) {
    var json = JSON.stringify(obj);
    var out  = callback ? callback + "(" + json + ")" : json;
    return ContentService.createTextOutput(out)
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }

  // Next serial number: highest SlNo in sheet + 1
  if (action === "getSlNo") {
    var maxSl = 0;
    for (var i = 1; i < rows.length; i++) {
      var v = parseInt(rows[i][1]);
      if (!isNaN(v) && v > maxSl) maxSl = v;
    }
    return respond({status:"ok", nextSl: maxSl + 1});
  }

  // Monthly rows filtered by ReceiptDate (col F)
  if (action === "getMonth") {
    var ym  = e.parameter.ym;
    var out = [];
    for (var i = 1; i < rows.length; i++) {
      var receiptISO = toISO_(rows[i][5]); // filter by ReceiptDate
      if (receiptISO.startsWith(ym)) {
        out.push([
          toISO_(rows[i][0]), // Date
          rows[i][1],          // SlNo
          rows[i][2],          // PatientName
          rows[i][3],          // Services
          rows[i][4],          // Fees
          receiptISO,          // ReceiptDate (ISO)
          rows[i][6],          // PaymentMode
          rows[i][7]           // Notes
        ]);
      }
    }
    return respond({status:"ok", rows: out});
  }

  return respond({status:"ok"});
}

// ── Optional: manually trigger a full sort from the Sheet menu ──
function sortSheetNow() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Form3C");
  sortByReceiptDate_(sheet);
  SpreadsheetApp.getUi().alert("✅ Sheet sorted by Receipt Date!");
}

⚠️ Sheet columns (A→J): ReceiptDate | SlNo | PatientName | UHID | Services | Fees | PaymentNature | PaymentMode | Notes | SubmittedAt

🔃 The sheet is automatically sorted by Receipt Date (ascending) after every new entry. Within the same receipt date, rows are ordered by Sl. No.

💡 You can also run sortSheetNow() manually from Apps Script anytime to re-sort the entire existing sheet.
